EXPLAIN (ANALYZE, BUFFERS, FORMAT TEXT)
WITH unfulfilled_demands AS (
    SELECT dr.material_id, COUNT(*) AS total_unfulfilled_demands,
           SUM(dr.quantity_required) AS total_demanded_quantity,
           ROUND(AVG(dr.max_unit_budget_inr), 2) AS avg_buyer_budget_inr,
           MIN(dr.required_by_date) AS earliest_deadline,
           COUNT(*) FILTER (WHERE dr.required_by_date <= CURRENT_DATE + INTERVAL '14 days') AS urgent_demands_count
    FROM demand_requirements dr
    WHERE dr.fulfillment_status IN ('OPEN','PARTIALLY_MATCHED') AND dr.required_by_date >= CURRENT_DATE
    GROUP BY dr.material_id
),
available_supplies AS (
    SELECT bl.material_id, COUNT(*) AS total_active_listings,
           SUM(bl.quantity_available) AS total_available_supply,
           ROUND(AVG(bl.price_per_unit_inr), 2) AS avg_seller_price_inr
    FROM byproduct_listings bl
    WHERE bl.listing_status = 'AVAILABLE' AND bl.expires_at >= CURRENT_TIMESTAMP
    GROUP BY bl.material_id
)
SELECT mc.material_code, mc.common_name, ud.total_demanded_quantity, asup.total_available_supply
FROM unfulfilled_demands ud
JOIN material_catalog mc ON ud.material_id = mc.material_id
LEFT JOIN available_supplies asup ON ud.material_id = asup.material_id
ORDER BY (ud.total_demanded_quantity - COALESCE(asup.total_available_supply,0)) DESC;
