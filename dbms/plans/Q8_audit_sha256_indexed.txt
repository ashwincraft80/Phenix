EXPLAIN (ANALYZE, BUFFERS, FORMAT TEXT)
WITH ordered_audit AS (
    SELECT al.log_id, al.entity_type, al.entity_id, al.event_type,
           al.actor_org_id, al.event_timestamp, al.payload_sha256,
           al.chained_hash,
           LAG(al.chained_hash) OVER (
               PARTITION BY al.entity_type, al.entity_id
               ORDER BY al.event_timestamp ASC
           ) AS prev_chained_hash,
           ROW_NUMBER() OVER (
               PARTITION BY al.entity_type, al.entity_id
               ORDER BY al.event_timestamp ASC
           ) AS event_sequence
    FROM audit_log al
    WHERE al.entity_type = 'EXCHANGE_TRANSACTION'
),
integrity_check AS (
    SELECT oa.*,
           CASE
               WHEN oa.event_sequence = 1 THEN 'GENESIS_BLOCK_VALID'
               WHEN oa.chained_hash = encode(
                   sha256((oa.prev_chained_hash || oa.payload_sha256)::bytea), 'hex'
               ) THEN 'CHAIN_INTEGRITY_VERIFIED'
               ELSE 'TAMPER_DETECTED_CHAIN_BROKEN'
           END AS chain_verification_status
    FROM ordered_audit oa
)
SELECT ic.log_id, ic.entity_id, ic.event_type, ic.event_timestamp,
       ic.payload_sha256, ic.chained_hash, ic.chain_verification_status
FROM integrity_check ic
WHERE ic.chain_verification_status != 'CHAIN_INTEGRITY_VERIFIED'
   OR ic.event_sequence = 1
ORDER BY ic.entity_id, ic.event_timestamp ASC;
