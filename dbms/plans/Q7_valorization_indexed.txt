EXPLAIN (ANALYZE, BUFFERS, FORMAT TEXT)
WITH completed_trades AS (
    SELECT et.buyer_org_id, et.seller_org_id, et.material_id,
           SUM(et.agreed_quantity) AS total_volume_traded_tonnes,
           SUM(et.agreed_quantity * et.agreed_unit_price_inr) AS total_secondary_spend_inr,
           SUM(et.agreed_quantity * mc.virgin_feedstock_benchmark_inr) AS imputed_virgin_cost_inr,
           SUM(et.agreed_quantity * 1.2) AS avoided_landfill_weight_inr
    FROM exchange_transactions et
    JOIN material_catalog mc ON et.material_id = mc.material_id
    WHERE et.transaction_status = 'COMPLETED'
      AND et.completed_at >= CURRENT_TIMESTAMP - INTERVAL '12 months'
    GROUP BY et.buyer_org_id, et.seller_org_id, et.material_id
)
SELECT org.legal_name, SUM(ct.imputed_virgin_cost_inr - ct.total_secondary_spend_inr) AS net_savings_inr
FROM completed_trades ct
JOIN organizations org ON ct.buyer_org_id = org.org_id
GROUP BY org.legal_name
ORDER BY net_savings_inr DESC;
