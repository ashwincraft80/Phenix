SET enable_indexscan = OFF;
SET enable_bitmapscan = OFF;
EXPLAIN (ANALYZE, BUFFERS, FORMAT TEXT)
WITH ordered_audit AS (
    SELECT al.log_id, al.entity_type, ..., LAG(al.chained_hash) OVER (...) AS prev_chained_hash,
           ROW_NUMBER() OVER (...) AS event_sequence
    FROM audit_log al WHERE al.entity_type = 'EXCHANGE_TRANSACTION'
),
integrity_check AS (
    SELECT oa.*, CASE ... WHEN sha256_recompute_matches THEN 'CHAIN_INTEGRITY_VERIFIED' ... END
    FROM ordered_audit oa
)
SELECT ... FROM integrity_check ic
WHERE ic.chain_verification_status != 'CHAIN_INTEGRITY_VERIFIED' OR ic.event_sequence = 1
ORDER BY ic.entity_id, ic.event_timestamp ASC;
