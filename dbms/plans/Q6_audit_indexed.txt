EXPLAIN (ANALYZE, BUFFERS, FORMAT TEXT)
SELECT lt.transit_id, lt.transit_reference, et.transaction_reference,
       mc.common_name AS material_name,
       ROUND((lt.weighbridge_net_kg - (et.agreed_quantity * 1000.00)), 2) AS mass_variance_kg,
       ROUND(((lt.weighbridge_net_kg - (et.agreed_quantity * 1000.00)) / NULLIF(et.agreed_quantity * 1000.00, 0.00) * 100.00)::numeric, 2) AS variance_pct
FROM logistics_transits lt
JOIN exchange_transactions et ON lt.transaction_id = et.transaction_id
JOIN material_catalog mc ON et.material_id = mc.material_id
JOIN organizations carrier ON lt.carrier_org_id = carrier.org_id
JOIN facilities sf ON et.seller_facility_id = sf.facility_id
JOIN facilities bf ON et.buyer_facility_id = bf.facility_id
WHERE ABS((lt.weighbridge_net_kg - (et.agreed_quantity * 1000.00)) / NULLIF(et.agreed_quantity * 1000.00, 0.00)) > 0.02
   OR lt.transit_status = 'DISCREPANCY_FLAGGED'
ORDER BY ABS(variance_pct) DESC;
